    input rst_n,
    input valid_in,
    input [7:0] data_in,

    output reg [15:0] data_out

reg [7:0] data_lock;
reg valid_lock;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        data_out <= 16'b0;
        valid_out <= 0;
        data_lock <= 8'b0;
        valid_lock <= 0;

            end else begin
